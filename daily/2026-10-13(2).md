        unordered_set<string> wordSet(wordList.begin(), wordList.end());
        if(wordSet.find(endWord) == wordSet.end()) return 0;
        

        unordered_set<string> beginSet, endSet, visitSet;
        beginSet.insert(beginWord);
        endSet.insert(endWord);
        visitSet.insert(beginWord);
        visitSet.insert(endWord);

        int step = 1;

        while (!beginSet.empty() && !endSet.empty()) {
            if (beginSet.size() > endSet.size()) {
                swap(beginSet, endSet);
            }
            unordered_set<string> newSet;

            for (string word : beginSet) {
                for (int i = 0; i < word.size(); i++) {
                    string newWord = word;
                    for (char c = 'a'; c <= 'z'; c++) {
                        newWord[i] = c;
                        if(endSet.find(newWord) != endSet.end()) {
                            return step + 1;
                        }
                        if(wordSet.find(newWord) != wordSet.end() && visitSet.find(newWord) == visitSet.end()){
                            visitSet.insert(newWord);
                            newSet.insert(newWord);
                        }
                    }
                }
            }
            beginSet = newSet;
            step++;
        }
        return 0;
    }
};
