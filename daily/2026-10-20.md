class Solution {
public:
    pair<int, int> test(vector<int>& nums, int start, int end) {
        bool drop = false;
        int mi = nums[start], ma = nums[start];
        for (int i = start + 1; i < end; i++) {
            if (nums[i] >= nums[i - 1]) ma = max(ma, nums[i]);
            else {
                if (!drop) {
                    drop = true;
                    mi = min(mi, nums[i]);
                }
                else return {-1, -1};
            }
        }
        if ((drop && nums[end - 1] <= nums[start]) || !drop) return {mi, ma};
        return {-1, -1};
    }

    int sortableIntegers(vector<int>& nums) {
        int ans = 0;
        int n = nums.size();
        for (int k = 1; k <= n; k++) {
            if (n % k == 0) {
                pair<int, int> p = {0, 0};
                bool flag = true;
                for (int i = 0; i < n; i+= k) {
                    pair<int, int> c = test(nums, i, i + k);
                    if (c.first < p.second) {
                        flag = false;
                        break;
                    } else p = c;
                }
                if (flag) ans += k;
            }
        }
        return ans;
    }
};
