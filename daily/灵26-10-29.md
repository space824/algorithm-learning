class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        int n = nums.size();
        double sum = 0;
        double maxAve = INT_MIN;
        for (int i = 0; i < n; i++) {
            sum += nums[i];

            int left = i - k + 1;
            if (left < 0) {
                continue;
            }

            double ave = sum / k;
            maxAve = max(maxAve, ave);

            sum -= nums[left];
        }
        return maxAve;
    }
};
