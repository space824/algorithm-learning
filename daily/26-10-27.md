class Solution {
public:
    int countStaircase(vector<int>& nums) {
        int n = nums.size();
        unordered_map<int, vector<int>> pos;
        vector<int> idx_in_pos(n);
        for (int i = 0; i < n; i++) {
            auto& ps = pos[nums[i]];
            idx_in_pos[i] = ps.size();
            ps.push_back(i);
        }

        vector<int> firsts;
        unordered_map<int, int> first_pos;
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            int x = nums[i];
            first_pos[x] = i;

            auto it = ranges::find(firsts, x);
            if (it != firsts.end()) {
                firsts.erase(it);
            }
            firsts.insert(firsts.begin(), x);
            int m = firsts.size();
            if (m * (m + 1) / 2 > n) {
                firsts.pop_back();
            }

            int mx = 0, mn = n;
            int r = i;
            for (int cnt = 1; cnt <= firsts.size(); cnt++) {
                x = firsts[cnt - 1];
                r += cnt;
                auto& ps = pos[x];
                int k = idx_in_pos[first_pos[x]] + cnt - 1;
                if (k >= ps.size()) {
                    break;
                }
                if (k + 1 < ps.size()) {
                    mn = min(mn, ps[k + 1]);
                }
                if (r >mn) {
                    break;
                }
                mx = max(mx, ps[k]);
                if (mx < r) {
                    ans++;
                }
            }
        }
        return ans;
    }
};
