class Solution {
public:
    int numOfSubarrays(vector<int>& arr, int k, int threshold) {
        int n = arr.size();
        int sum = 0, cnt = 0;

        for (int i = 0; i < n; i++) {
            sum += arr[i];

            int left = i - k + 1;
            if (left < 0) {
                continue;
            }

            int ave = sum / k;
            if (ave >= threshold) {
                cnt++;
            }
            sum -= arr[left];
        }
        return cnt;
    }
};

class Solution {
public:
    vector<int> getAverages(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> result(n, -1);
        long long sum = 0;

        for (int i = 0; i < n; i++) {
            sum += nums[i];

            int left = i - 2 * k;
            if (left < 0) {
                continue;
            }

            int ave = sum / (2 * k + 1);
            
            result[i - k] = ave;

            sum -= nums[left];
        }
        return result;
    }
};

class Solution {
public:
    int minimumRecolors(string blocks, int k) {
        int n = blocks.size();
        int sum = 0;
        int maxSum = INT_MIN;

        for (int i = 0; i < n; i++) {
            if (blocks[i] == 'B') {
                sum++;
            }

            int left = i - k + 1;
            if (left < 0) {
                continue;
            }

            maxSum = max(maxSum, sum);

            if (blocks[left] == 'B') {
                sum--;
            }
        }
        return k - maxSum;
    }
};
