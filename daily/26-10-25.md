void floyd(vector<vector<ll>>& d) {
    int n = d.size();
    for (int k = 0; k < n; ++k)                  // k 必须在最外层
        for (int i = 0; i < n; ++i) {
            if (d[i][k] == INF) continue;
            for (int j = 0; j < n; ++j) {
                if (d[k][j] == INF) continue;    // 防止 INF + INF 溢出
                if (d[i][k] + d[k][j] < d[i][j])
                    d[i][j] = d[i][k] + d[k][j];
            }
        }
}
