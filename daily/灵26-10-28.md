class Solution {
public:
    int maxVowels(string s, int k) {
        int ans = 0, vowel = 0;
        for (int i = 0; i < s.size(); i++) { //枚举右端点
            if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u') {
                vowel++;
            }

            int left = i - k + 1;
            if (left < 0) {
                continue;
            }

            ans = max(ans, vowel);
            if (ans == k) {
                break;    // 优化剪枝
            }

            char c = s[left];
            if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
                vowel--;
            }
        }
        return ans;
    }
};
