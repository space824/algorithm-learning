#include <iostream>
#include <vector>
#include <climits>

using namespace std;

int main() {
    int v, e;
    int x, y, k;
    cin >> v >> e;

    vector<vector<int>> grid(v + 1, vector<int>(v + 1, INT_MAX));
    while (e--) {
        cin >> x >> y >> k;
        grid[x][y] = k;
        grid[y][x] = k;
    }

    vector<int> minDist(v + 1, INT_MAX);
    vector<bool> isIntree(v + 1, false);

    minDist[1] = 0;

    for (int i = 1; i <= v; i++) {
        int cur = -1;
        int minVal = INT_MAX;
        for (int j = 1; j <= v; j++) {
            if (!isIntree[j] && minDist[j] < minVal) {
                minVal = minDist[j];
                cur = j;
            }
        }

        if (cur == -1) break;

        isIntree[cur] = true;

        for (int j = 1; j <= v; j++) {
            if (!isIntree[j] && grid[cur][j] < minDist[j]) {
                minDist[j] = grid[cur][j];
            }
        }
    }

    int result = 0;
    for (int i = 2; i <= v; i++) {
        result += minDist[i];
    }
    cout << result << endl;
}
