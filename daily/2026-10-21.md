    static const int N = 500000;
    vector<int> father;

    void init() {
        father.resize(N);
        for (int i = 0; i < N; i++) {
            father[i] = i;
        }
    }

    int find(int u) {
        if (u == father[u]) return u;
        return father[u] = find(father[u]);
    }

    void join(int u, int v) {
        u = find(u);
        v = find(v);
        if (u == v) return;
        else father[v] = u;
    }

    bool isSame(int u, int v) {
        u = find(u);
        v = find(v);
        return u == v;
    }

    struct Edge {
        int start;
        int end;
        int val;
    };

    static bool cmp(Edge a, Edge b) {
        return a.val < b.val;
    }

    int kruskal(vector<vector<int>>& points) {
        init();
        int result = 0;
        vector<Edge> edges;
        int n = points.size();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                Edge s = {i, j, abs(points[i][0] - points[j][0]) + abs(points[i][1] - points[j][1])};
                edges.push_back(s);
            }
        }
        sort(edges.begin(), edges.end(), cmp);
        int m = edges.size();
        for (Edge edge : edges) {
            if (!isSame(edge.start, edge.end)) {
                join(edge.start, edge.end);
                result += edge.val;
            }
        }
        return result;
    }

    int minCostConnectPoints(vector<vector<int>>& points) {
        //return prim(points, 0);
        return kruskal(points);
    }
};
