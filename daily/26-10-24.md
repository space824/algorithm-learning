// 返回 false 表示存在从 src 可达的负环
bool bellmanFord(int n, const vector<Edge>& edges, int src, vector<ll>& dist) {
    const ll INF = (1LL << 62);
    dist.assign(n, INF);
    dist[src] = 0;

    for (int i = 0; i < n - 1; ++i) {
        bool changed = false;
        for (const auto& e : edges) {
            if (dist[e.u] == INF) continue;      // 不判会溢出
            if (dist[e.u] + e.w < dist[e.v]) {
                dist[e.v] = dist[e.u] + e.w;
                changed = true;
            }
        }
        if (!changed) return true;               // 提前收敛
    }

    for (const auto& e : edges)                  // 第 n 轮
        if (dist[e.u] != INF && dist[e.u] + e.w < dist[e.v]) return false;
    return true;
}
